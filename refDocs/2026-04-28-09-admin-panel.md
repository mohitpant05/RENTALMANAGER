# Admin Panel Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add backend API endpoints for tenant administration — theme configuration, tenant settings, staff/user management, domain management, and platform superadmin tenant oversight.

**Architecture:** Two router modules: `app/core/admin/` for tenant-scoped admin operations (theme, staff, domains, tenant config) and a platform-level router for superadmin tenant CRUD. A `TenantTheme` model stores per-tenant theming in public schema. Admin endpoints use `require_permission("theme:*:edit")` or `user.user_type == "admin"` checks. Platform superadmin uses the existing `PlatformUser` model for auth (separate JWT flow without tenant context).

**Tech Stack:** FastAPI, SQLModel, Pydantic v2

import uuid
from datetime import datetime, timezone

from sqlalchemy import Column, DateTime
from sqlalchemy.dialects.postgresql import JSONB
from sqlmodel import Field, SQLModel


class TenantTheme(SQLModel, table=True):
    __tablename__ = "tenant_themes"

    id: uuid.UUID = Field(default_factory=uuid.uuid4, primary_key=True)
    tenant_id: uuid.UUID = Field(foreign_key="tenants.id", unique=True, index=True)
    brand: dict = Field(default_factory=dict, sa_column=Column(JSONB, nullable=False))
    colors: dict = Field(default_factory=dict, sa_column=Column(JSONB, nullable=False))
    typography: dict = Field(
        default_factory=dict, sa_column=Column(JSONB, nullable=False)
    )
    components: dict = Field(
        default_factory=dict, sa_column=Column(JSONB, nullable=False)
    )
    features: dict = Field(
        default_factory=dict, sa_column=Column(JSONB, nullable=False)
    )
    updated_at: datetime = Field(
        default_factory=lambda: datetime.now(timezone.utc),
        sa_column=Column(DateTime(timezone=True), nullable=False),
    )

# backend/app/core/admin/service.py
import uuid
from datetime import datetime, timezone

from fastapi import HTTPException
from sqlmodel import select
from sqlmodel.ext.asyncio.session import AsyncSession

from app.core.admin.models import TenantTheme
from app.core.admin.schemas import DomainCreate, StaffInvite, ThemeUpdate
from app.core.auth.models import Role, User, UserRole
from app.core.auth.password import hash_password
from app.core.tenancy.models import Tenant, TenantDomain


async def get_theme(session: AsyncSession, tenant_id: uuid.UUID) -> TenantTheme:
    result = await session.exec(
        select(TenantTheme).where(TenantTheme.tenant_id == tenant_id)
    )
    theme = result.first()
    if theme is None:
        theme = TenantTheme(tenant_id=tenant_id)
        session.add(theme)
        await session.commit()
        await session.refresh(theme)
    return theme


async def update_theme(
    session: AsyncSession, tenant_id: uuid.UUID, data: ThemeUpdate
) -> TenantTheme:
    theme = await get_theme(session, tenant_id)
    if data.brand is not None:
        theme.brand = data.brand
    if data.colors is not None:
        theme.colors = data.colors
    if data.typography is not None:
        theme.typography = data.typography
    if data.components is not None:
        theme.components = data.components
    if data.features is not None:
        theme.features = data.features
    theme.updated_at = datetime.now(timezone.utc)
    session.add(theme)
    await session.commit()
    await session.refresh(theme)
    return theme


async def update_tenant_config(
    session: AsyncSession,
    tenant: Tenant,
    name: str | None = None,
    base_currency: str | None = None,
    tz: str | None = None,
    config: dict | None = None,
) -> Tenant:
    if name is not None:
        tenant.name = name
    if base_currency is not None:
        tenant.base_currency = base_currency
    if tz is not None:
        tenant.timezone = tz
    if config is not None:
        tenant.config = config
    tenant.updated_at = datetime.now(timezone.utc)
    session.add(tenant)
    await session.commit()
    await session.refresh(tenant)
    return tenant


async def list_domains(
    session: AsyncSession, tenant_id: uuid.UUID
) -> list[TenantDomain]:
    result = await session.exec(
        select(TenantDomain).where(TenantDomain.tenant_id == tenant_id)
    )
    return list(result.all())


async def add_domain(
    session: AsyncSession, tenant_id: uuid.UUID, data: DomainCreate
) -> TenantDomain:
    result = await session.exec(
        select(TenantDomain).where(TenantDomain.domain == data.domain)
    )
    if result.first() is not None:
        raise HTTPException(status_code=409, detail="Domain already registered")
    domain = TenantDomain(
        tenant_id=tenant_id,
        domain=data.domain,
        is_primary=data.is_primary,
    )
    session.add(domain)
    await session.commit()
    await session.refresh(domain)
    return domain


async def remove_domain(
    session: AsyncSession, tenant_id: uuid.UUID, domain_id: uuid.UUID
) -> None:
    result = await session.exec(
        select(TenantDomain).where(
            TenantDomain.id == domain_id, TenantDomain.tenant_id == tenant_id
        )
    )
    domain = result.first()
    if domain is None:
        raise HTTPException(status_code=404, detail="Domain not found")
    await session.delete(domain)
    await session.commit()


async def list_staff(session: AsyncSession) -> list[dict]:
    result = await session.exec(
        select(User).where(User.user_type.in_(["admin", "staff"]))
    )
    users = list(result.all())
    staff_list = []
    for u in users:
        roles_result = await session.exec(
            select(Role.slug)
            .join(UserRole, UserRole.role_id == Role.id)
            .where(UserRole.user_id == u.id)
        )
        roles = list(roles_result.all())
        staff_list.append({**u.model_dump(), "roles": roles})
    return staff_list


async def invite_staff(session: AsyncSession, data: StaffInvite) -> User:
    result = await session.exec(select(User).where(User.email == data.email))
    if result.first() is not None:
        raise HTTPException(status_code=409, detail="Email already registered")

    user = User(
        email=data.email,
        password_hash=hash_password(data.password),
        full_name=data.full_name,
        user_type=data.user_type,
    )
    session.add(user)
    await session.commit()
    await session.refresh(user)

    for slug in data.role_slugs:
        role_result = await session.exec(select(Role).where(Role.slug == slug))
        role = role_result.first()
        if role is not None:
            user_role = UserRole(user_id=user.id, role_id=role.id)
            session.add(user_role)
    await session.commit()
    return user


async def deactivate_user(session: AsyncSession, user_id: uuid.UUID) -> User:
    result = await session.exec(select(User).where(User.id == user_id))
    user = result.first()
    if user is None:
        raise HTTPException(status_code=404, detail="User not found")
    user.is_active = False
    user.updated_at = datetime.now(timezone.utc)
    session.add(user)
    await session.commit()
    await session.refresh(user)
    return user
```

# backend/app/core/admin/platform_auth.py
from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer
from jose import JWTError, jwt
from sqlmodel import select
from sqlmodel.ext.asyncio.session import AsyncSession

from app.config import get_settings
from app.core.auth.models import PlatformUser
from app.core.auth.password import verify_password
from app.database import get_session

_security = HTTPBearer()


def create_platform_token(user_id: str) -> str:
    settings = get_settings()
    payload = {
        "sub": user_id,
        "type": "platform",
    }
    return jwt.encode(payload, settings.jwt_secret_key, algorithm=settings.jwt_algorithm)


async def authenticate_platform_user(
    session: AsyncSession, email: str, password: str
) -> PlatformUser:
    result = await session.exec(
        select(PlatformUser).where(PlatformUser.email == email)
    )
    user = result.first()
    if user is None or not verify_password(password, user.password_hash):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid credentials",
        )
    if not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Account deactivated",
        )
    return user


async def get_current_platform_user(
    credentials: HTTPAuthorizationCredentials = Depends(_security),
    session: AsyncSession = Depends(get_session),
) -> PlatformUser:
    settings = get_settings()
    try:
        payload = jwt.decode(
            credentials.credentials,
            settings.jwt_secret_key,
            algorithms=[settings.jwt_algorithm],
        )
    except JWTError:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token",
        )
    if payload.get("type") != "platform":
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Not a platform token",
        )
    user_id = payload.get("sub")
    result = await session.exec(
        select(PlatformUser).where(PlatformUser.id == user_id)
    )
    user = result.first()
    if user is None or not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="User not found or inactive",
        )
    return user
```

# backend/app/core/admin/platform_router.py
import uuid

from fastapi import APIRouter, Depends, HTTPException
from sqlmodel import select
from sqlmodel.ext.asyncio.session import AsyncSession

from app.core.admin.platform_auth import (
    authenticate_platform_user,
    create_platform_token,
    get_current_platform_user,
)
from app.core.admin.schemas import (
    PlatformLoginRequest,
    PlatformTokenResponse,
    TenantCreate,
    TenantResponse,
)
from app.core.auth.models import PlatformUser
from app.core.tenancy.models import Tenant
from app.core.tenancy.provisioning import provision_tenant_schema
from app.database import engine, get_session

router = APIRouter(prefix="/api/platform", tags=["platform"])


@router.post("/login", response_model=PlatformTokenResponse)
async def platform_login(
    data: PlatformLoginRequest,
    session: AsyncSession = Depends(get_session),
):
    user = await authenticate_platform_user(session, data.email, data.password)
    token = create_platform_token(str(user.id))
    return PlatformTokenResponse(access_token=token)


@router.get("/tenants", response_model=list[TenantResponse])
async def list_tenants(
    session: AsyncSession = Depends(get_session),
    platform_user: PlatformUser = Depends(get_current_platform_user),
):
    result = await session.exec(select(Tenant))
    tenants = list(result.all())
    return [TenantResponse.model_validate(t) for t in tenants]


@router.post("/tenants", response_model=TenantResponse, status_code=201)
async def create_tenant(
    data: TenantCreate,
    session: AsyncSession = Depends(get_session),
    platform_user: PlatformUser = Depends(get_current_platform_user),
):
    result = await session.exec(select(Tenant).where(Tenant.slug == data.slug))
    if result.first() is not None:
        raise HTTPException(status_code=409, detail="Slug already taken")

    tenant_id = uuid.uuid4()
    schema_name = f"tenant_{str(tenant_id).replace('-', '_')}"
    tenant = Tenant(
        id=tenant_id,
        name=data.name,
        slug=data.slug,
        schema_name=schema_name,
        base_currency=data.base_currency,
        timezone=data.timezone,
    )
    session.add(tenant)
    await session.commit()
    await session.refresh(tenant)

    await provision_tenant_schema(engine, session, str(tenant_id))
    return TenantResponse.model_validate(tenant)


@router.get("/tenants/{tenant_id}", response_model=TenantResponse)
async def get_tenant(
    tenant_id: uuid.UUID,
    session: AsyncSession = Depends(get_session),
    platform_user: PlatformUser = Depends(get_current_platform_user),
):
    result = await session.exec(select(Tenant).where(Tenant.id == tenant_id))
    tenant = result.first()
    if tenant is None:
        raise HTTPException(status_code=404, detail="Tenant not found")
    return TenantResponse.model_validate(tenant)


@router.post("/tenants/{tenant_id}/deactivate", response_model=TenantResponse)
async def deactivate_tenant(
