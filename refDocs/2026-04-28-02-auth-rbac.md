# Auth & RBAC Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build tenant-scoped authentication (register, login, JWT access+refresh tokens) and role-based access control (custom roles, granular permissions with resource-level scoping) for the RentManager platform.

**Architecture:** Password-based auth with bcrypt hashing, JWT access tokens (15min) + Redis-backed refresh tokens (7 days, revocable, rotated on use). RBAC with permission format `<domain>:<resource_qualifier>:<action>` and wildcard matching. User hierarchy: admin (implicit all-permissions) → staff (role-based) → customer → lister. All user/role/permission data lives in tenant schemas. Platform superadmin uses a separate public-schema table.

**Tech Stack:** FastAPI + SQLModel + python-jose (JWT) + passlib (bcrypt) + Redis (refresh tokens) + PostgreSQL (schema-per-tenant)

**Spec reference:** `docs/superpowers/specs/2026-04-28-rental-management-platform-design.md` — "Auth & RBAC" section (lines 257-304)

```python
from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncEngine, AsyncSession
from sqlmodel import SQLModel

from app.core.auth.models import TENANT_AUTH_TABLES  # noqa: F401 — registers models

TENANT_TABLE_REGISTRY: set[str] = set()
TENANT_TABLE_REGISTRY.update(TENANT_AUTH_TABLES)


async def create_tenant_tables(engine: AsyncEngine, schema_name: str) -> None:
    tables = [
        t for t in SQLModel.metadata.sorted_tables if t.name in TENANT_TABLE_REGISTRY
    ]
    async with engine.begin() as conn:
        await conn.execute(text(f"SET search_path TO {schema_name}, public"))
        for table in tables:
            await conn.run_sync(lambda sync_conn, t=table: t.create(sync_conn, checkfirst=True))
        await conn.execute(text("SET search_path TO public"))


async def provision_tenant_schema(engine: AsyncEngine, session: AsyncSession, tenant_id: str) -> str:
    schema_name = f"tenant_{tenant_id.replace('-', '_')}"
    await session.execute(text(f"CREATE SCHEMA IF NOT EXISTS {schema_name}"))
    await session.commit()
    await create_tenant_tables(engine, schema_name)
    return schema_name
```

```python
import uuid
from datetime import datetime, timedelta, timezone

from jose import JWTError, jwt

from app.config import get_settings
from app.redis_client import get_redis


def create_access_token(user_id: str, tenant_id: str, roles: list[str]) -> str:
    settings = get_settings()
    expire = datetime.now(timezone.utc) + timedelta(minutes=settings.access_token_expire_minutes)
    payload = {
        "sub": user_id,
        "tenant_id": tenant_id,
        "roles": roles,
        "type": "access",
        "exp": expire,
        "jti": str(uuid.uuid4()),
    }
    return jwt.encode(payload, settings.secret_key, algorithm=settings.jwt_algorithm)


def create_refresh_token(user_id: str, tenant_id: str) -> tuple[str, str]:
    settings = get_settings()
    jti = str(uuid.uuid4())
    expire = datetime.now(timezone.utc) + timedelta(days=settings.refresh_token_expire_days)
    payload = {
        "sub": user_id,
        "tenant_id": tenant_id,
        "type": "refresh",
        "exp": expire,
        "jti": jti,
    }
    token = jwt.encode(payload, settings.secret_key, algorithm=settings.jwt_algorithm)
    return token, jti


def decode_token(token: str) -> dict:
    settings = get_settings()
    return jwt.decode(token, settings.secret_key, algorithms=[settings.jwt_algorithm])


async def store_refresh_token(jti: str, user_id: str, tenant_id: str) -> None:
    settings = get_settings()
    r = get_redis()
    ttl = settings.refresh_token_expire_days * 86400
    await r.setex(f"refresh:{jti}", ttl, f"{user_id}:{tenant_id}")


async def is_refresh_token_valid(jti: str) -> bool:
    r = get_redis()
    return await r.exists(f"refresh:{jti}") > 0


async def revoke_refresh_token(jti: str) -> None:
    r = get_redis()
    await r.delete(f"refresh:{jti}")
```

from datetime import datetime, timezone

from fastapi import HTTPException, status
from sqlalchemy.ext.asyncio import AsyncSession
from sqlmodel import select

from app.core.auth.jwt import (
    create_access_token,
    create_refresh_token,
    decode_token,
    is_refresh_token_valid,
    revoke_refresh_token,
    store_refresh_token,
)
from app.core.auth.models import Role, User, UserRole
from app.core.auth.password import hash_password, verify_password
from app.core.auth.schemas import (
    LoginRequest,
    RefreshRequest,
    RegisterRequest,
    TokenResponse,
    UserResponse,
)


async def register_user(
    session: AsyncSession,
    data: RegisterRequest,
    tenant_id: str,
    user_type: str = "customer",
) -> User:
    result = await session.exec(select(User).where(User.email == data.email))
    if result.first() is not None:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="Email already registered",
        )

    user = User(
        email=data.email,
        password_hash=hash_password(data.password),
        full_name=data.full_name,
        phone=data.phone,
        user_type=user_type,
    )
    session.add(user)
    await session.commit()
    await session.refresh(user)
    return user


async def authenticate_user(session: AsyncSession, data: LoginRequest) -> User:
    result = await session.exec(select(User).where(User.email == data.email))
    user = result.first()
    if user is None or not verify_password(data.password, user.password_hash):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid email or password",
        )
    if not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Account is deactivated",
        )

    user.last_login = datetime.now(timezone.utc)
    session.add(user)
    await session.commit()
    return user


async def get_user_roles(session: AsyncSession, user_id: str) -> list[str]:
    result = await session.exec(
        select(Role.slug)
        .join(UserRole, UserRole.role_id == Role.id)
        .where(UserRole.user_id == user_id)
    )
    return list(result.all())


async def create_tokens(user: User, tenant_id: str, roles: list[str]) -> TokenResponse:
    access_token = create_access_token(
        user_id=str(user.id),
        tenant_id=tenant_id,
        roles=roles,
    )
    refresh_token, jti = create_refresh_token(
        user_id=str(user.id),
        tenant_id=tenant_id,
    )
    await store_refresh_token(jti, str(user.id), tenant_id)
    return TokenResponse(access_token=access_token, refresh_token=refresh_token)


async def refresh_tokens(session: AsyncSession, data: RefreshRequest) -> TokenResponse:
    try:
        payload = decode_token(data.refresh_token)
    except Exception:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid refresh token",
        )

    if payload.get("type") != "refresh":
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token type",
        )

    jti = payload.get("jti")
    if not await is_refresh_token_valid(jti):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Refresh token revoked",
        )

    await revoke_refresh_token(jti)

    user_id = payload["sub"]
    tenant_id = payload["tenant_id"]

    result = await session.exec(select(User).where(User.id == user_id))
    user = result.first()
    if user is None or not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="User not found or inactive",
        )

    roles = await get_user_roles(session, str(user.id))
    return await create_tokens(user, tenant_id, roles)


def build_user_response(user: User, roles: list[str]) -> UserResponse:
    return UserResponse(
        id=user.id,
        email=user.email,
        full_name=user.full_name,
        phone=user.phone,
        is_active=user.is_active,
        is_verified=user.is_verified,
        user_type=user.user_type,
        avatar_url=user.avatar_url,
        created_at=user.created_at,
        roles=roles,
    )
```

import uuid

from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer
from jose import JWTError
from sqlalchemy.ext.asyncio import AsyncSession
from sqlmodel import select

from app.core.auth.jwt import decode_token
from app.core.auth.models import Permission, RolePermission, User, UserRole
from app.core.tenancy.context import get_current_tenant
from app.database import get_db

_security = HTTPBearer()


def matches_permission(required: str, granted: set[str]) -> bool:
    if required in granted:
        return True
    parts = required.split(":")
    if len(parts) != 3:
        return False
    domain, qualifier, action = parts
    candidates = [
        f"{domain}:*:{action}",
        f"{domain}:{qualifier}:*",
        f"{domain}:*:*",
        f"*:*:*",
    ]
    return any(c in granted for c in candidates)


async def get_user_permissions(session: AsyncSession, user_id: uuid.UUID) -> set[str]:
    result = await session.exec(
        select(Permission.codename)
        .join(RolePermission, RolePermission.permission_id == Permission.id)
        .join(UserRole, UserRole.role_id == RolePermission.role_id)
        .where(UserRole.user_id == user_id)
    )
    return set(result.all())


async def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(_security),
    session: AsyncSession = Depends(get_db),
) -> User:
    try:
        payload = decode_token(credentials.credentials)
    except JWTError:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token",
        )

    if payload.get("type") != "access":
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token type",
        )

    tenant_ctx = get_current_tenant()
    if tenant_ctx is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="No tenant context",
        )

    if payload.get("tenant_id") != str(tenant_ctx.tenant.id):
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Token tenant mismatch",
        )

    user_id = payload.get("sub")
    result = await session.exec(select(User).where(User.id == user_id))
    user = result.first()
    if user is None or not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="User not found or inactive",
        )
    return user


def require_permission(permission: str):
    async def _checker(
        user: User = Depends(get_current_user),
        session: AsyncSession = Depends(get_db),
    ) -> User:
        if user.user_type == "admin":
            return user

        user_perms = await get_user_permissions(session, user.id)
        if not matches_permission(permission, user_perms):
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Insufficient permissions",
            )
        return user

    return _checker
```

from fastapi import APIRouter, Depends, status
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.auth.dependencies import get_current_user
from app.core.auth.models import User
from app.core.auth.schemas import (
    LoginRequest,
    RefreshRequest,
    RegisterRequest,
    TokenResponse,
    UserResponse,

    authenticate_user,
    build_user_response,
    create_tokens,
    get_user_roles,
    refresh_tokens,
    register_user,
)
from app.core.auth.jwt import decode_token, revoke_refresh_token
from app.core.tenancy.context import get_current_tenant
from app.database import get_db

router = APIRouter(prefix="/api/auth", tags=["auth"])


@router.post("/register", response_model=TokenResponse, status_code=status.HTTP_201_CREATED)
async def register(
    data: RegisterRequest,
    session: AsyncSession = Depends(get_db),
):
    tenant_ctx = get_current_tenant()
    tenant_id = str(tenant_ctx.tenant.id)
    user = await register_user(session, data, tenant_id=tenant_id)
    roles = await get_user_roles(session, str(user.id))
    return await create_tokens(user, tenant_id, roles)


@router.post("/login", response_model=TokenResponse)
async def login(
    data: LoginRequest,
    session: AsyncSession = Depends(get_db),
):
    tenant_ctx = get_current_tenant()
    tenant_id = str(tenant_ctx.tenant.id)
    user = await authenticate_user(session, data)
    roles = await get_user_roles(session, str(user.id))
    return await create_tokens(user, tenant_id, roles)


@router.post("/refresh", response_model=TokenResponse)
async def refresh(
    data: RefreshRequest,
    session: AsyncSession = Depends(get_db),
):
    return await refresh_tokens(session, data)


@router.post("/logout")
async def logout(
    data: RefreshRequest,
    user: User = Depends(get_current_user),
):
    try:
        payload = decode_token(data.refresh_token)
        jti = payload.get("jti")
        if jti:
            await revoke_refresh_token(jti)
    except Exception:
        pass
    return {"detail": "Logged out"}


@router.get("/me", response_model=UserResponse)
async def me(
    user: User = Depends(get_current_user),
    session: AsyncSession = Depends(get_db),
):
    roles = await get_user_roles(session, str(user.id))
    return build_user_response(user, roles)

async def seed_default_roles(session: AsyncSession) -> None:
    perm_map: dict[str, Permission] = {}
    all_codenames = {code for code, _ in DEFAULT_PERMISSIONS}
    all_codenames.add("*:*:*")

    for codename, description in [*DEFAULT_PERMISSIONS, ("*:*:*", "Superuser — all permissions")]:
        result = await session.exec(
            select(Permission).where(Permission.codename == codename)
        )
        existing = result.first()
        if existing:
            perm_map[codename] = existing
        else:
            perm = Permission(codename=codename, description=description)
            session.add(perm)
            perm_map[codename] = perm

    await session.flush()

    for slug, config in DEFAULT_ROLES.items():
        result = await session.exec(select(Role).where(Role.slug == slug))
        existing_role = result.first()
        if existing_role:
            continue

        role = Role(
            name=config["name"],
            slug=slug,
            description=config["description"],
            is_system=True,
        )
        session.add(role)
        await session.flush()

        for perm_codename in config["permissions"]:
            perm = perm_map.get(perm_codename)
            if perm:
                rp = RolePermission(role_id=role.id, permission_id=perm.id)
                session.add(rp)

    await session.commit()
