**Architecture:** A single Python package (`rm_mcp/`) at project root. Indexer pipeline parses Python files with stdlib `ast` and TypeScript/TSX files with `tree-sitter`. Results stored as a single JSON index at `.rm-mcp/index.json`. MCP server exposes 8 tools + 5 resource URIs over stdio. Git post-commit hook triggers background reindex.

**Tech Stack:** Python 3.11+, `fastmcp` (MCP SDK), `tree-sitter` + `tree-sitter-typescript` (frontend parsing), stdlib `ast` (backend parsing)

## Global Constraints

- Python 3.11+ (match backend runtime)
- No external services, no API keys, no database — pure local
- `.rm-mcp/` directory is gitignored
- Module name `rm_mcp` — invoked as `python -m rm_mcp`
- All file paths in index are relative to project root
- Use `fastmcp` package (v2+), not raw `mcp` SDK

# rm_mcp/indexer/python_parser.py
import ast
from pathlib import Path


def _rel(path: Path, root: Path) -> str:
    return str(path.relative_to(root))


def parse_router_file(filepath: Path, project_root: Path) -> list[dict]:
    source = filepath.read_text()
    tree = ast.parse(source, filename=str(filepath))
    routes = []
    prefix = ""

    for node in ast.walk(tree):
        # Find APIRouter(prefix=...) call
        if isinstance(node, ast.Assign):
            for target in node.targets:
                if isinstance(target, ast.Name) and isinstance(node.value, ast.Call):
                    for kw in node.value.keywords:
                        if kw.arg == "prefix" and isinstance(kw.value, ast.Constant):
                            prefix = kw.value.value

    for node in ast.walk(tree):
        if not isinstance(node, ast.AsyncFunctionDef) and not isinstance(node, ast.FunctionDef):
            continue
        for deco in node.decorator_list:
            if not isinstance(deco, ast.Call):
                continue
            if not isinstance(deco.func, ast.Attribute):
                continue
            method = deco.func.attr.upper()
            if method not in ("GET", "POST", "PUT", "DELETE", "PATCH"):
                continue

            path_arg = ""
            if deco.args and isinstance(deco.args[0], ast.Constant):
                path_arg = deco.args[0].value

            full_path = prefix + path_arg

            deps = []
            service_calls = []
            for arg in node.args.args + node.args.kwonlyargs:
                pass
            for default in node.args.defaults + node.args.kw_defaults:
                if default is None:
                    continue
                if isinstance(default, ast.Call) and isinstance(default.func, ast.Name):
                    if default.func.id == "Depends":
                        if default.args and isinstance(default.args[0], ast.Name):
                            deps.append(default.args[0].id)
                        elif default.args and isinstance(default.args[0], ast.Call):
                            if isinstance(default.args[0].func, ast.Name):
                                deps.append(default.args[0].func.id + "(...)")
                            elif isinstance(default.args[0].func, ast.Attribute):
                                deps.append(default.args[0].func.attr + "(...)")

            # Extract service calls from function body
            for body_node in ast.walk(node):
                if isinstance(body_node, ast.Await) and isinstance(body_node.value, ast.Call):
                    call = body_node.value
                    if isinstance(call.func, ast.Name):
                        service_calls.append(call.func.id)
                    elif isinstance(call.func, ast.Attribute):
                        service_calls.append(call.func.attr)

            auth_required = any("current_user" in d or "permission" in d for d in deps)

            routes.append({
                "method": method,
                "path": full_path,
                "handler": node.name,
                "file": _rel(filepath, project_root),
                "line": node.lineno,
                "service_calls": service_calls,
                "auth_required": auth_required,
                "dependencies": deps,
            })

    return routes


def parse_model_file(filepath: Path, project_root: Path) -> list[dict]:
    source = filepath.read_text()
    tree = ast.parse(source, filename=str(filepath))
    models = []

    for node in ast.iter_child_nodes(tree):
        if not isinstance(node, ast.ClassDef):
            continue

        is_table_model = False
        for kw in node.keywords:
            if kw.arg == "table" and isinstance(kw.value, ast.Constant) and kw.value.value is True:
                is_table_model = True

        if not is_table_model:
            continue

        table_name = ""
        columns = []
        relationships = []

        for item in node.body:
            if isinstance(item, ast.Assign):
                for target in item.targets:
                    if isinstance(target, ast.Name) and target.id == "__tablename__":
                        if isinstance(item.value, ast.Constant):
                            table_name = item.value.value
            elif isinstance(item, ast.AnnAssign) and isinstance(item.target, ast.Name):
                col_name = item.target.id
                col_type = ""
                if isinstance(item.annotation, ast.Name):
                    col_type = item.annotation.id
                elif isinstance(item.annotation, ast.Subscript):
                    if isinstance(item.annotation.value, ast.Name):
                        col_type = item.annotation.value.id
                elif isinstance(item.annotation, ast.BinOp):
                    col_type = "Optional"
                elif isinstance(item.annotation, ast.Attribute):
                    col_type = item.annotation.attr

                # Check for foreign_key in Field() — marks a relationship
                is_fk = False
                if item.value and isinstance(item.value, ast.Call):
                    for kw in item.value.keywords:
                        if kw.arg == "foreign_key" and isinstance(kw.value, ast.Constant):
                            fk_table = kw.value.value.split(".")[0]
                            if fk_table not in relationships:
                                relationships.append(fk_table)
                            is_fk = True

                columns.append({"name": col_name, "type": col_type, "is_fk": is_fk})

        models.append({
            "name": node.name,
            "table": table_name,
            "file": _rel(filepath, project_root),
            "line": node.lineno,
            "columns": columns,
            "relationships": relationships,
        })

    return models


def parse_schema_file(filepath: Path, project_root: Path) -> list[dict]:
    source = filepath.read_text()
    tree = ast.parse(source, filename=str(filepath))
    schemas = []

    for node in ast.iter_child_nodes(tree):
        if not isinstance(node, ast.ClassDef):
            continue
        is_basemodel = any(
            (isinstance(base, ast.Name) and base.id == "BaseModel")
            or (isinstance(base, ast.Attribute) and base.attr == "BaseModel")
            for base in node.bases
        )
        if not is_basemodel:
            continue

        fields = []
        for item in node.body:
            if isinstance(item, ast.AnnAssign) and isinstance(item.target, ast.Name):
                field_name = item.target.id
                field_type = ast.dump(item.annotation) if item.annotation else "Any"
                # Simplify type representation
                if isinstance(item.annotation, ast.Name):
                    field_type = item.annotation.id
                elif isinstance(item.annotation, ast.Subscript):
                    if isinstance(item.annotation.value, ast.Name):
                        field_type = f"{item.annotation.value.id}[...]"
                elif isinstance(item.annotation, ast.BinOp):
                    field_type = "Optional"
                elif isinstance(item.annotation, ast.Attribute):
                    field_type = item.annotation.attr

                has_default = item.value is not None
                fields.append({"name": field_name, "type": field_type, "has_default": has_default})

        schemas.append({
            "name": node.name,
            "file": _rel(filepath, project_root),
            "line": node.lineno,
            "fields": fields,
        })

    return schemas


def parse_service_file(filepath: Path, project_root: Path) -> list[dict]:
    source = filepath.read_text()
    tree = ast.parse(source, filename=str(filepath))
    services = []

    module_name = filepath.parent.name

    for node in ast.iter_child_nodes(tree):
        if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            continue
        if node.name.startswith("_"):
            continue

        params = []
        for arg in node.args.args:
            if arg.arg == "self":
                continue
            p_type = ""
            if arg.annotation:
                if isinstance(arg.annotation, ast.Name):
                    p_type = arg.annotation.id
                elif isinstance(arg.annotation, ast.Attribute):
                    p_type = arg.annotation.attr
                elif isinstance(arg.annotation, ast.Subscript):
                    if isinstance(arg.annotation.value, ast.Name):
                        p_type = f"{arg.annotation.value.id}[...]"
            params.append({"name": arg.arg, "type": p_type})

        # Find model references and function calls in body
        models_used = []
        calls = []
        for body_node in ast.walk(node):
            if isinstance(body_node, ast.Name) and body_node.id[0].isupper():
                if body_node.id not in models_used and body_node.id not in (
                    "HTTPException", "AsyncSession", "None", "True", "False",
                    "Optional", "UUID", "Path", "Query",
                ):
                    models_used.append(body_node.id)
            if isinstance(body_node, ast.Call):
                if isinstance(body_node.func, ast.Attribute) and body_node.func.attr == "delay":
                    if isinstance(body_node.func.value, ast.Name):
                        calls.append(f"{body_node.func.value.id}.delay")

        services.append({
            "name": node.name,
            "module": module_name,
            "file": _rel(filepath, project_root),
            "line": node.lineno,
            "params": params,
            "models_used": models_used,
            "calls": calls,
            "is_async": isinstance(node, ast.AsyncFunctionDef),
        })

    return services


def parse_celery_tasks(filepath: Path, project_root: Path) -> list[dict]:
    source = filepath.read_text()
    tree = ast.parse(source, filename=str(filepath))
    tasks = []

    for node in ast.iter_child_nodes(tree):
        if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            continue
        for deco in node.decorator_list:
            if not isinstance(deco, ast.Call):
                continue
            if not isinstance(deco.func, ast.Attribute):
                continue
            if deco.func.attr != "task":
                continue

            task_name = node.name
            celery_name = ""
            for kw in deco.keywords:
                if kw.arg == "name" and isinstance(kw.value, ast.Constant):
                    celery_name = kw.value.value

            tasks.append({
                "name": celery_name or task_name,
                "function": task_name,
                "file": _rel(filepath, project_root),
                "line": node.lineno,
            })

    return tasks


def parse_backend(backend_dir: Path, project_root: Path) -> dict:
    routes = []
    services = []
    models = []
    schemas = []
    tasks = []

    for py_file in backend_dir.rglob("*.py"):
        if "__pycache__" in str(py_file):
            continue
        name = py_file.name

        try:
            if name.startswith("router") and name.endswith(".py"):
                routes.extend(parse_router_file(py_file, project_root))
            elif name == "service.py":
                services.extend(parse_service_file(py_file, project_root))
            elif name == "models.py":
                models.extend(parse_model_file(py_file, project_root))
            elif name == "schemas.py":
                schemas.extend(parse_schema_file(py_file, project_root))
        except SyntaxError:
            continue

    # Parse celery tasks from celery_app.py
    celery_file = project_root / "backend" / "app" / "celery_app.py"
    if celery_file.exists():
        try:
            tasks.extend(parse_celery_tasks(celery_file, project_root))
        except SyntaxError:
            pass

    return {
        "routes": routes,
        "services": services,
        "models": models,
        "schemas": schemas,
        "tasks": tasks,
    }

# rm_mcp/indexer/ts_parser.py
import re
from pathlib import Path


def _rel(path: Path, root: Path) -> str:
    return str(path.relative_to(root))


def _route_from_page_path(filepath: Path, frontend_dir: Path) -> str:
    """Convert file path to Next.js route.

    frontend/src/app/admin/bookings/page.tsx -> /admin/bookings
    frontend/src/app/(public)/resources/[id]/page.tsx -> /(public)/resources/[id]
    """
    rel = filepath.relative_to(frontend_dir / "app")
    parts = list(rel.parts[:-1])  # drop page.tsx
    if not parts:
        return "/"
    return "/" + "/".join(parts)


def extract_api_calls(source: str) -> list[dict]:
    """Extract api.get/post/put/delete calls using regex.

    Matches patterns like:
      api.post<Type>("/bookings", payload)
      api.get(`/resources/${id}`)
      api.post<void>(`/resources/${resourceId}/availability/release`, ...)
    """
    pattern = r'api\.(get|post|put|delete|patch)\s*(?:<[^>]*>)?\s*\(\s*[`"\']([^`"\']+)[`"\']'
    calls = []
    for i, line in enumerate(source.splitlines(), 1):
        for match in re.finditer(pattern, line):
            method = match.group(1).upper()
            path = match.group(2)
            # Normalize template literal variables
            path = re.sub(r'\$\{[^}]+\}', '*', path)
            calls.append({"method": method, "path": path, "line": i})
    return calls


def extract_imports(source: str) -> list[dict]:
    """Extract import statements to find hook and component usage."""
    pattern = r'import\s+(?:{([^}]+)}|(\w+))\s+from\s+["\']([^"\']+)["\']'
    imports = []
    for match in re.finditer(pattern, source):
        named = match.group(1)
        default = match.group(2)
        module = match.group(3)
        if named:
            names = [n.strip().split(" as ")[0].strip() for n in named.split(",")]
            for name in names:
                if name:
                    imports.append({"name": name, "module": module})
        if default:
            imports.append({"name": default, "module": module})
    return imports


def parse_pages(frontend_dir: Path, project_root: Path) -> list[dict]:
    pages = []
    app_dir = frontend_dir / "app"

    for page_file in app_dir.rglob("page.tsx"):
        route = _route_from_page_path(page_file, frontend_dir)
        source = page_file.read_text()

        imports = extract_imports(source)
        hooks_used = [
            imp["name"] for imp in imports
            if imp["name"].startswith("use") and imp["name"][3:4].isupper()
        ]
        components_used = [
            imp["name"] for imp in imports
            if imp["name"][0:1].isupper()
            and not imp["name"].startswith("use")
            and imp["module"].startswith("@/components")
        ]
        api_calls = extract_api_calls(source)

        pages.append({
            "route": route,
            "file": _rel(page_file, project_root),
            "hooks_used": hooks_used,
            "components_used": components_used,
            "api_calls": api_calls,
        })

    return pages


def parse_hooks(frontend_dir: Path, project_root: Path) -> list[dict]:
    hooks = []
    hooks_dir = frontend_dir / "hooks"

    if not hooks_dir.exists():
        return hooks

    for hook_file in hooks_dir.glob("use-*.ts"):
        source = hook_file.read_text()

        # Derive hook name: use-booking-flow.ts -> useBookingFlow
        stem = hook_file.stem  # use-booking-flow
        parts = stem.split("-")
        hook_name = parts[0] + "".join(p.capitalize() for p in parts[1:])

        api_calls = extract_api_calls(source)
        imports = extract_imports(source)

        stores_used = [
            imp["name"] for imp in imports
            if "store" in imp["module"].lower()
        ]

        hooks.append({
            "name": hook_name,
            "file": _rel(hook_file, project_root),
            "api_calls": api_calls,
            "stores_used": stores_used,
        })

    return hooks


def parse_stores(frontend_dir: Path, project_root: Path) -> list[dict]:
    stores = []
    lib_dir = frontend_dir / "lib"

    if not lib_dir.exists():
        return stores

    for store_file in lib_dir.glob("*-store.ts"):
        source = store_file.read_text()

        # Derive store name: auth-store.ts -> authStore
        stem = store_file.stem  # auth-store
        parts = stem.split("-")
        store_name = parts[0] + "".join(p.capitalize() for p in parts[1:])

        # Extract exported function/interface names
        exports = re.findall(r'export\s+(?:function|const|interface|type)\s+(\w+)', source)

        stores.append({
            "name": store_name,
            "file": _rel(store_file, project_root),
            "exports": exports,
        })

    return stores


def parse_components(frontend_dir: Path, project_root: Path) -> list[dict]:
    components = []
    comp_dir = frontend_dir / "components"

    if not comp_dir.exists():
        return components

    for tsx_file in comp_dir.rglob("*.tsx"):
        if tsx_file.name == "page.tsx" or tsx_file.name == "layout.tsx":
            continue

        source = tsx_file.read_text()
        exports = re.findall(r'export\s+(?:default\s+)?function\s+(\w+)', source)
        if not exports:
            exports = re.findall(r'export\s+const\s+(\w+)', source)

        imports = extract_imports(source)
        hooks_used = [
            imp["name"] for imp in imports
            if imp["name"].startswith("use") and imp["name"][3:4].isupper()
        ]

        for export_name in exports:
            components.append({
                "name": export_name,
                "file": _rel(tsx_file, project_root),
                "hooks_used": hooks_used,
            })

    return components


def parse_frontend(frontend_dir: Path, project_root: Path) -> dict:
    return {
        "pages": parse_pages(frontend_dir, project_root),
        "hooks": parse_hooks(frontend_dir, project_root),
        "stores": parse_stores(frontend_dir, project_root),
        "components": parse_components(frontend_dir, project_root),
    }
```

> **Note:** This uses regex-based parsing instead of tree-sitter. The regex approach is simpler, has zero native dependencies, and handles the actual API call patterns in this codebase (`api.post<Type>("/path", ...)`, template literals). Tree-sitter can be swapped in later if regex proves insufficient, but for ~120 TS/TSX files with consistent patterns, regex is reliable and dependency-free.

# rm_mcp/indexer/linker.py
import re


def _normalize_path(path: str) -> str:
    """Normalize API path for comparison.

    /api/bookings/{id}/confirm -> bookings/*/confirm
    /resources/*/availability/check -> resources/*/availability/check
    """
    path = re.sub(r'^/api/?', '', path)
    path = re.sub(r'\{[^}]+\}', '*', path)
    path = path.strip('/')
    return path


def match_api_path(frontend_path: str, backend_path: str) -> bool:
    front_norm = _normalize_path(frontend_path)
    back_norm = _normalize_path(backend_path)

    if front_norm == back_norm:
        return True

    # Try matching with wildcard expansion
    front_parts = front_norm.split('/')
    back_parts = back_norm.split('/')

    if len(front_parts) != len(back_parts):
        return False

    for fp, bp in zip(front_parts, back_parts):
        if fp == '*' or bp == '*':
            continue
        if fp != bp:
            return False

    return True


def _find_service_for_handler(handler_name: str, service_calls: list[str], services: list[dict]) -> dict | None:
    for call_name in service_calls:
        for svc in services:
            if svc["name"] == call_name:
                return svc
    return None


def _find_model_for_service(service: dict, models: list[dict]) -> str | None:
    for model_name in service.get("models_used", []):
        return model_name
    return None


def build_links(backend: dict, frontend: dict) -> list[dict]:
    links = []

    # Build page -> hook mapping
    page_hooks = {}
    for page in frontend["pages"]:
        for hook_name in page["hooks_used"]:
            page_hooks.setdefault(hook_name, []).append(page)

    # For each hook's API call, find the matching backend route
    for hook in frontend["hooks"]:
        for api_call in hook["api_calls"]:
            for route in backend["routes"]:
                if api_call["method"] != route["method"]:
                    continue
                if not match_api_path(api_call["path"], route["path"]):
                    continue

                service = _find_service_for_handler(
                    route["handler"], route["service_calls"], backend["services"]
                )
                model_name = _find_model_for_service(service, backend["models"]) if service else None

                pages_using_hook = page_hooks.get(hook["name"], [])
                for page in pages_using_hook:
                    links.append({
                        "page": page["route"],
                        "hook": hook["name"],
                        "api_path": f"{route['method']} {route['path']}",
                        "handler": f"{route['file'].split('/')[-2]}.router.{route['handler']}",
                        "service": f"{service['module']}.service.{service['name']}" if service else None,
                        "model": model_name,
                    })

    # Also check pages with direct API calls (no hook)
    for page in frontend["pages"]:
        for api_call in page["api_calls"]:
            for route in backend["routes"]:
                if api_call["method"] != route["method"]:
                    continue
                if not match_api_path(api_call["path"], route["path"]):
                    continue

                service = _find_service_for_handler(
                    route["handler"], route["service_calls"], backend["services"]
                )
                model_name = _find_model_for_service(service, backend["models"]) if service else None

                links.append({
                    "page": page["route"],
                    "hook": None,
                    "api_path": f"{route['method']} {route['path']}",
                    "handler": f"{route['file'].split('/')[-2]}.router.{route['handler']}",
                    "service": f"{service['module']}.service.{service['name']}" if service else None,
                    "model": model_name,
                })

    return links
```

# rm_mcp/indexer/pipeline.py
import json
import sys
from datetime import UTC, datetime
from pathlib import Path

from rm_mcp.config import BACKEND_DIR, FRONTEND_DIR, INDEX_DIR, INDEX_PATH, PROJECT_ROOT
from rm_mcp.indexer.linker import build_links

def run_reindex(quiet: bool = False) -> dict:
    if not quiet:
        print("Indexing backend...")
    backend = parse_backend(BACKEND_DIR, PROJECT_ROOT)

    if not quiet:
        print("Indexing frontend...")
    frontend = parse_frontend(FRONTEND_DIR, PROJECT_ROOT)

    if not quiet:
        print("Building cross-stack links...")
    links = build_links(backend, frontend)

    index = {
        "version": 1,
        "indexed_at": datetime.now(UTC).isoformat(),
        "backend": backend,
        "frontend": frontend,
        "links": links,
    }

    INDEX_DIR.mkdir(parents=True, exist_ok=True)
    INDEX_PATH.write_text(json.dumps(index, indent=2, default=str))

    if not quiet:
        print(f"Index built: {len(backend['routes'])} routes, "
              f"{len(backend['models'])} models, "
              f"{len(frontend['pages'])} pages, "
              f"{len(frontend['hooks'])} hooks, "
              f"{len(links)} links")
        print(f"Saved to {INDEX_PATH}")

    return index


def load_index() -> dict:
    if not INDEX_PATH.exists():
        return {}
    return json.loads(INDEX_PATH.read_text())


def show_status():
    if not INDEX_PATH.exists():
        print("No index found. Run: python -m rm_mcp reindex")
        return

    index = load_index()
    indexed_at = index.get("indexed_at", "unknown")
    backend = index.get("backend", {})
    frontend = index.get("frontend", {})
    links = index.get("links", [])

    print(f"Indexed at:    {indexed_at}")
    print(f"Routes:        {len(backend.get('routes', []))}")
    print(f"Services:      {len(backend.get('services', []))}")
    print(f"Models:        {len(backend.get('models', []))}")
    print(f"Schemas:       {len(backend.get('schemas', []))}")
    print(f"Celery tasks:  {len(backend.get('tasks', []))}")
    print(f"Pages:         {len(frontend.get('pages', []))}")
    print(f"Hooks:         {len(frontend.get('hooks', []))}")
    print(f"Stores:        {len(frontend.get('stores', []))}")
    print(f"Components:    {len(frontend.get('components', []))}")
    print(f"Links:         {len(links)}")

# rm_mcp/query.py


def trace_flow(index: dict, from_page: str) -> list[dict]:
    """Trace full path from page to backend for matching links."""
    links = index.get("links", [])
    matches = []

    from_page = from_page.strip("/")
    for link in links:
        page_clean = link["page"].strip("/")
        if from_page in page_clean or page_clean in from_page:
            matches.append(link)

    if not matches:
        # Try matching by hook name
        for link in links:
            if link.get("hook") and from_page.lower() in link["hook"].lower():
                matches.append(link)

    if not matches:
        # Try matching by route path
        for link in links:
            if from_page.lower() in link.get("api_path", "").lower():
                matches.append(link)

    return matches


def list_routes(index: dict, module: str | None = None) -> list[dict]:
    routes = index.get("backend", {}).get("routes", [])
    if module:
        module_lower = module.lower()
        routes = [r for r in routes if module_lower in r["path"].lower() or module_lower in r["file"].lower()]
    return routes


def list_pages(index: dict, area: str | None = None) -> list[dict]:
    pages = index.get("frontend", {}).get("pages", [])
    if area:
        area_lower = area.lower()
        pages = [p for p in pages if area_lower in p["route"].lower()]
    return pages


def find_symbol(index: dict, name: str) -> list[dict]:
    results = []
    name_lower = name.lower()
    backend = index.get("backend", {})
    frontend = index.get("frontend", {})

    for route in backend.get("routes", []):
        if name_lower in route["handler"].lower():
            results.append({**route, "type": "route"})

    for svc in backend.get("services", []):
        if name_lower in svc["name"].lower():
            results.append({**svc, "type": "service"})

    for model in backend.get("models", []):
        if name_lower in model["name"].lower():
            results.append({**model, "type": "model"})

    for schema in backend.get("schemas", []):
        if name_lower in schema["name"].lower():
            results.append({**schema, "type": "schema"})

    for task in backend.get("tasks", []):
        if name_lower in task["name"].lower() or name_lower in task.get("function", "").lower():
            results.append({**task, "type": "task"})

    for hook in frontend.get("hooks", []):
        if name_lower in hook["name"].lower():
            results.append({**hook, "type": "hook"})

    for page in frontend.get("pages", []):
        if name_lower in page["route"].lower() or name_lower in page["file"].lower():
            results.append({**page, "type": "page"})

    for store in frontend.get("stores", []):
        if name_lower in store["name"].lower():
            results.append({**store, "type": "store"})

    return results


def query_index(index: dict, q: str) -> list[dict]:
    """Fuzzy search across all indexed items and knowledge docs."""
    results = []
    q_lower = q.lower()
    terms = q_lower.split()

    def score(text: str) -> int:
        text_lower = text.lower()
        return sum(1 for t in terms if t in text_lower)

    backend = index.get("backend", {})
    frontend = index.get("frontend", {})

    for route in backend.get("routes", []):
        s = score(f"{route['path']} {route['handler']} {route['file']}")
        if s > 0:
            results.append({"score": s, "type": "route", "item": route})

    for svc in backend.get("services", []):
        s = score(f"{svc['name']} {svc['module']} {svc['file']}")
        if s > 0:
            results.append({"score": s, "type": "service", "item": svc})

    for model in backend.get("models", []):
        s = score(f"{model['name']} {model['table']} {model['file']}")
        if s > 0:
            results.append({"score": s, "type": "model", "item": model})

    for schema in backend.get("schemas", []):
        s = score(f"{schema['name']} {schema['file']}")
        if s > 0:
            results.append({"score": s, "type": "schema", "item": schema})

    for task in backend.get("tasks", []):
        s = score(f"{task['name']} {task.get('function', '')} {task['file']}")
        if s > 0:
            results.append({"score": s, "type": "task", "item": task})

    for page in frontend.get("pages", []):
        s = score(f"{page['route']} {page['file']}")
        if s > 0:
            results.append({"score": s, "type": "page", "item": page})

    for hook in frontend.get("hooks", []):
        s = score(f"{hook['name']} {hook['file']}")
        if s > 0:
            results.append({"score": s, "type": "hook", "item": hook})

    results.sort(key=lambda r: r["score"], reverse=True)
    return results[:20]


def get_schema(index: dict, name: str) -> dict | None:
    name_lower = name.lower()

    for model in index.get("backend", {}).get("models", []):
        if model["name"].lower() == name_lower:
            return {**model, "kind": "model"}

    for schema in index.get("backend", {}).get("schemas", []):
        if schema["name"].lower() == name_lower:
            return {**schema, "kind": "schema"}

    return None

# rm_mcp/docs.py
from rm_mcp.config import KNOWLEDGE_DIR


def list_docs() -> list[dict]:
    docs = []
    if not KNOWLEDGE_DIR.exists():
        return docs
    for md_file in sorted(KNOWLEDGE_DIR.glob("*.md")):
        topic = md_file.stem
        first_line = ""
        with open(md_file) as f:
            for line in f:
                line = line.strip()
                if line and not line.startswith("#"):
                    first_line = line[:120]
                    break
                elif line.startswith("# "):
                    first_line = line[2:].strip()
                    break
        docs.append({
            "topic": topic,
            "file": md_file.name,
            "title": first_line,
        })
    return docs


def get_doc(topic: str) -> str | None:
    md_file = KNOWLEDGE_DIR / f"{topic}.md"
    if md_file.exists():
        return md_file.read_text()

    # Try partial match
    if KNOWLEDGE_DIR.exists():
        for f in KNOWLEDGE_DIR.glob("*.md"):
            if topic.lower() in f.stem.lower():
                return f.read_text()

    return None


def search_docs(q: str) -> list[dict]:
    results = []
    q_lower = q.lower()
    terms = q_lower.split()

    if not KNOWLEDGE_DIR.exists():
        return results

    for md_file in KNOWLEDGE_DIR.glob("*.md"):
        content = md_file.read_text()
        content_lower = content.lower()
        score = sum(content_lower.count(t) for t in terms)
        if score > 0:
            # Extract first matching paragraph
            snippet = ""
            for para in content.split("\n\n"):
                if any(t in para.lower() for t in terms):
                    snippet = para.strip()[:200]
                    break
            results.append({
                "topic": md_file.stem,
                "score": score,
                "snippet": snippet,
            })

    results.sort(key=lambda r: r["score"], reverse=True)
    return results[:10]
```

# rm_mcp/server.py
import json

from fastmcp import FastMCP

from rm_mcp.config import INDEX_PATH
from rm_mcp.docs import get_doc as _get_doc, list_docs, search_docs
from rm_mcp.indexer.pipeline import load_index
from rm_mcp.query import (
    find_symbol as _find_symbol,
    get_schema as _get_schema,
    list_pages as _list_pages,
    list_routes as _list_routes,
    query_index,
    trace_flow as _trace_flow,
)

mcp = FastMCP(
    name="rm-mcp",
    version="1.0.0",
)


def _ensure_index() -> dict:
    index = load_index()
    if not index:
        from rm_mcp.indexer.pipeline import run_reindex
        run_reindex(quiet=True)
        index = load_index()
    return index


@mcp.tool()
def trace_flow(from_identifier: str) -> str:
    """Trace the full path from a frontend page or hook through API calls to backend service and database model.

    Args:
        from_identifier: Page route (e.g. "admin/bookings"), hook name (e.g. "useBookingFlow"), or API path (e.g. "/bookings")
    """
    index = _ensure_index()
    traces = _trace_flow(index, from_identifier)
    if not traces:
        return f"No traces found for '{from_identifier}'"
    return json.dumps(traces, indent=2)


@mcp.tool()
def list_routes(module: str | None = None) -> str:
    """List all backend API routes with methods, handlers, and auth requirements.

    Args:
        module: Optional filter by module name (e.g. "bookings", "payments"). Returns all routes if omitted.
    """
    index = _ensure_index()
    routes = _list_routes(index, module)
    if not routes:
        return f"No routes found" + (f" for module '{module}'" if module else "")
    return json.dumps(routes, indent=2)


@mcp.tool()
def list_pages(area: str | None = None) -> str:
    """List all frontend pages with their hooks and API calls.

    Args:
        area: Optional filter by area (e.g. "admin", "public", "superadmin"). Returns all pages if omitted.
    """
    index = _ensure_index()
    pages = _list_pages(index, area)
    if not pages:
        return f"No pages found" + (f" for area '{area}'" if area else "")
    return json.dumps(pages, indent=2)


@mcp.tool()
def find_symbol(name: str) -> str:
    """Find a function, class, model, hook, or component by name across the entire codebase.

    Args:
        name: Symbol name to search for (e.g. "create_booking", "Booking", "useBookingFlow")
    """
    index = _ensure_index()
    results = _find_symbol(index, name)
    if not results:
        return f"No symbol found matching '{name}'"
    return json.dumps(results, indent=2)


@mcp.tool()
def query(q: str) -> str:
    """Fuzzy search across all indexed items — routes, services, models, schemas, tasks, pages, hooks.

    Args:
        q: Search query (e.g. "payment processing", "booking confirm", "auth")
    """
    index = _ensure_index()
    results = query_index(index, q)

    # Also search docs
    doc_results = search_docs(q)
    for doc in doc_results:
        results.append({"score": doc["score"], "type": "doc", "item": doc})

    results.sort(key=lambda r: r["score"], reverse=True)
    if not results:
        return f"No results found for '{q}'"
    return json.dumps(results[:20], indent=2)


@mcp.tool()
def get_schema(name: str) -> str:
    """Get detailed schema for a Pydantic model or SQLAlchemy model — fields, types, relationships.

    Args:
        name: Model or schema name (e.g. "Booking", "BookingCreate", "User")
    """
    index = _ensure_index()
    result = _get_schema(index, name)
    if not result:
        return f"No schema found for '{name}'"
    return json.dumps(result, indent=2)


@mcp.tool()
def get_doc(topic: str) -> str:
    """Get a knowledge document by topic name.

    Args:
        topic: Document topic (e.g. "architecture-overview", "booking-state-machine", "auth-flow")
    """
    content = _get_doc(topic)
    if not content:
        available = list_docs()
        topics = [d["topic"] for d in available]
        return f"No doc found for '{topic}'. Available: {', '.join(topics)}"
    return content


@mcp.tool()
def overview() -> str:
    """Get a project overview: module counts, route counts, page counts, index freshness."""
    index = _ensure_index()
    backend = index.get("backend", {})
    frontend = index.get("frontend", {})
    links = index.get("links", [])
    docs = list_docs()

    summary = {
        "indexed_at": index.get("indexed_at"),
        "backend": {
            "routes": len(backend.get("routes", [])),
            "services": len(backend.get("services", [])),
            "models": len(backend.get("models", [])),
            "schemas": len(backend.get("schemas", [])),
            "celery_tasks": len(backend.get("tasks", [])),
        },
        "frontend": {
            "pages": len(frontend.get("pages", [])),
            "hooks": len(frontend.get("hooks", [])),
            "stores": len(frontend.get("stores", [])),
            "components": len(frontend.get("components", [])),
        },
        "cross_stack_links": len(links),
        "knowledge_docs": len(docs),
        "doc_topics": [d["topic"] for d in docs],
    }
    return json.dumps(summary, indent=2)


# --- Resources ---

@mcp.resource("rm://overview")
def resource_overview() -> str:
    """Project summary — module list, route count, page count."""
    return overview()


@mcp.resource("rm://routes")
def resource_routes() -> str:
    """Complete API route table."""
    return list_routes()


@mcp.resource("rm://pages")
def resource_pages() -> str:
    """Complete page → endpoint mapping."""
    return list_pages()


@mcp.resource("rm://docs/{topic}")
def resource_doc(topic: str) -> str:
    """Individual knowledge doc by topic name."""
    return get_doc(topic)


@mcp.resource("rm://index-status")
def resource_index_status() -> str:
    """Index age, file counts, staleness."""
    index = _ensure_index()
    return json.dumps({
        "indexed_at": index.get("indexed_at"),
        "version": index.get("version"),
        "index_path": str(INDEX_PATH),
    }, indent=2)


def serve():
    mcp.run(transport="stdio")
```
